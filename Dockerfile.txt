FROM python:3.11-slim

# Install FFmpeg so yt-dlp can merge 1080p+ video and convert audio formats
RUN apt-get update && \
    apt-get install -y --no-install-recommends ffmpeg && \
    rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
# Always pull the latest yt-dlp version so platform extractors stay updated
RUN pip install --no-cache-dir --upgrade -r requirements.txt yt-dlp

COPY . .

EXPOSE 10000

CMD ["gunicorn", "--bind", "0.0.0.0:10000", "--workers", "2", "--threads", "4", "--timeout", "600", "app:app"]
